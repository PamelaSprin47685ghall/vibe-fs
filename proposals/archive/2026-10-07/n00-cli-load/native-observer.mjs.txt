import fs from 'node:fs'
import path from 'node:path'
import { pathToFileURL } from 'node:url'
import { registerHooks } from 'node:module'

const root = '/Users/yuanxi/Workwork/vibe-fs'
const entry = path.join(root, 'scripts/migrate-delegation-history.mjs')
if (process.argv[1] && path.resolve(process.argv[1]) === entry) {
  const directory = process.env.WXS_MIGRATION_LOAD_OBSERVATIONS
  if (!directory || !path.isAbsolute(directory)) throw new Error('missing owned observation directory')
  const receipt = path.join(directory, `${process.pid}.ndjson`)
  const descriptor = fs.openSync(receipt, 'wx')
  fs.closeSync(descriptor)
  let sequence = 0
  const record = (phase, extra = {}) => {
    if (sequence >= 32) throw new Error('migration load observation exceeded its finite bound')
    fs.appendFileSync(receipt, `${JSON.stringify({
      phase, sequence: sequence++, pid: process.pid, parentPid: process.ppid,
      monotonicMs: performance.now(), cpu: process.cpuUsage(), ...extra,
    })}\n`)
  }
  const frontDoors = new Set([
    pathToFileURL(path.join(root, 'dist/Persistence/EventStore/Surface.js')).href,
    pathToFileURL(path.join(root, 'dist/Strength/Surface.js')).href,
  ])
  record('ready', { entry, argv: process.argv.slice(2) })
  registerHooks({
    load(url, context, nextLoad) {
      const loaded = nextLoad(url, context)
      if (frontDoors.has(url)) record('actual-next-load-complete', { url, format: loaded.format })
      return loaded
    },
  })
  process.on('exit', code => record('exit', { code }))
}
