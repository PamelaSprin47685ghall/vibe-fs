import assert from 'node:assert/strict'
import fs from 'node:fs'
import path from 'node:path'
import { syncBuiltinESMExports } from 'node:module'
import { pathToFileURL } from 'node:url'

const [mode, repositoryRoot, commonDir, writerId, scenario] = process.argv.slice(2)
assert.ok(['measure', 'cold'].includes(mode))
assert.ok(['fresh', 'duplicate', 'normal'].includes(scenario))
const incoming = {
  id: '7'.repeat(40), stream: 'proof/u0-release', type: 'JobRequested',
  parents: [], payload: { text: '真实释放之后-é', detail: { ready: true } }, payloadRefs: [],
}
const writerFile = path.join(commonDir, 'wanxiang', 'events', `${writerId}.ndjson`)
const lockPath = path.join(commonDir, 'wanxiang.lock')
const injected = new Error('owned native release completed before its response failed')
const original = {
  rmSync: fs.rmSync, appendFileSync: fs.appendFileSync, openSync: fs.openSync,
  closeSync: fs.closeSync, fsyncSync: fs.fsyncSync,
}
const descriptors = new Map()
let armed = false
let recording = false
const observed = { appendCalls: 0, fsyncCalls: 0, releaseCalls: 0, injectedCalls: 0 }
const filename = value => typeof value === 'string' ? path.resolve(value) : null

if (mode === 'measure') {
  fs.appendFileSync = function (...args) {
    if (recording && filename(args[0]) === writerFile) observed.appendCalls += 1
    return Reflect.apply(original.appendFileSync, this, args)
  }
  fs.openSync = function (...args) {
    const descriptor = Reflect.apply(original.openSync, this, args)
    descriptors.set(descriptor, filename(args[0]))
    return descriptor
  }
  fs.closeSync = function (...args) {
    try { return Reflect.apply(original.closeSync, this, args) }
    finally { descriptors.delete(args[0]) }
  }
  fs.fsyncSync = function (...args) {
    if (recording && descriptors.get(args[0]) === writerFile) observed.fsyncCalls += 1
    return Reflect.apply(original.fsyncSync, this, args)
  }
  fs.rmSync = function (...args) {
    const ownRelease = recording && filename(args[0]) === lockPath
    if (ownRelease) observed.releaseCalls += 1
    const result = Reflect.apply(original.rmSync, this, args)
    if (ownRelease && armed) {
      armed = false
      observed.injectedCalls += 1
      assert.equal(fs.existsSync(lockPath), false, 'original native removal actually completed')
      throw injected
    }
    return result
  }
  syncBuiltinESMExports()
}

let handle
try {
  const store = await import(pathToFileURL(path.join(repositoryRoot, 'dist/Persistence/EventStore/Surface.js')))
  const codec = await import(pathToFileURL(path.join(repositoryRoot, 'dist/Persistence/EventStore/CodecSurface.js')))
  handle = store.create(commonDir, writerId)
  if (mode === 'cold') {
    assert.deepEqual(store.read(handle, incoming.id), incoming)
    assert.equal(store.head(handle, incoming.stream), incoming.id)
    assert.deepEqual(store.heads(handle, incoming.stream), [incoming.id])
    process.stdout.write(JSON.stringify({ pid: process.pid, event: incoming, head: incoming.id, heads: [incoming.id] }))
  } else {
    if (scenario === 'duplicate') {
      assert.deepEqual(await store.append(handle, [incoming]), { ok: true, cuts: [] })
      assert.deepEqual(store.read(handle, incoming.id), incoming)
    }
    const before = fs.existsSync(writerFile) ? fs.readFileSync(writerFile) : Buffer.alloc(0)
    recording = true
    armed = scenario !== 'normal'
    let result
    let thrown
    try { result = await store.append(handle, [incoming]) }
    catch (error) { thrown = error }
    recording = false

    const bytes = fs.readFileSync(writerFile)
    const expectedBytes = Buffer.from(codec.encode(incoming), 'utf8')
    assert.deepEqual(bytes, expectedBytes, 'actual writer contains the complete canonical envelope plus LF')
    if (scenario === 'duplicate') assert.deepEqual(bytes, before, 'exact duplicate adds no bytes')
    assert.deepEqual(store.read(handle, incoming.id), incoming, 'actual canonical Current includes the fact')
    assert.equal(store.head(handle, incoming.stream), incoming.id)
    assert.deepEqual(store.heads(handle, incoming.stream), [incoming.id])
    assert.equal(fs.existsSync(lockPath), false, 'actual owned lock directory is absent')
    assert.equal(observed.releaseCalls, 1, 'the actual operation traversed native release once')
    assert.equal(observed.injectedCalls, scenario === 'normal' ? 0 : 1)
    assert.equal(observed.appendCalls, scenario === 'duplicate' ? 0 : 1)
    assert.equal(observed.fsyncCalls, scenario === 'duplicate' ? 0 : 1)
    process.stdout.write(JSON.stringify({
      pid: process.pid, scenario, observed, bytesBase64: bytes.toString('base64'),
      beforeBase64: before.toString('base64'), event: incoming, head: incoming.id, heads: [incoming.id],
      settled: thrown === undefined,
      thrownIsInjectedCause: thrown === injected,
      result: result ?? null,
      causeIsInjected: result?.error?.cause === injected,
    }))
  }
} finally {
  if (handle !== undefined) {
    const store = await import(pathToFileURL(path.join(repositoryRoot, 'dist/Persistence/EventStore/Surface.js')))
    store.dispose(handle)
  }
  Object.assign(fs, original)
  syncBuiltinESMExports()
}
