import assert from 'node:assert/strict'
import { readFileSync } from 'node:fs'

const scenario = process.argv[2]
assert.ok(['fresh', 'duplicate', 'normal'].includes(scenario))
const measured = JSON.parse(readFileSync(`/private/tmp/vibe-fs-u0-${scenario}-measure-20261006.json`, 'utf8'))
const cold = JSON.parse(readFileSync(`/private/tmp/vibe-fs-u0-${scenario}-cold-20261006.json`, 'utf8'))
assert.ok(Number.isInteger(measured.pid) && measured.pid > 0)
assert.ok(Number.isInteger(cold.pid) && cold.pid > 0 && cold.pid !== measured.pid)
assert.deepEqual(cold.event, measured.event)
assert.equal(cold.head, measured.head)
assert.deepEqual(cold.heads, measured.heads)
assert.ok(Buffer.from(measured.bytesBase64, 'base64').at(-1) === 10)
assert.equal(measured.observed.releaseCalls, 1)
assert.equal(measured.observed.appendCalls, scenario === 'duplicate' ? 0 : 1)
assert.equal(measured.observed.fsyncCalls, scenario === 'duplicate' ? 0 : 1)
if (scenario === 'duplicate') assert.equal(measured.beforeBase64, measured.bytesBase64)
process.stdout.write(JSON.stringify({
  oraclePid: process.pid, measurePid: measured.pid, coldPid: cold.pid, scenario,
  diskCurrentColdPositive: true, originalRejectionCause: measured.thrownIsInjectedCause,
}) + '\n')

assert.equal(measured.settled, true, 'actual Store append must settle its typed release outcome instead of rejecting its Task')
if (scenario === 'normal') {
  assert.deepEqual(measured.result, { ok: true, cuts: [] })
} else {
  const expectedCode = scenario === 'fresh' ? 'CommitUnknown' : 'NoNewWriteReleaseFailed'
  assert.equal(measured.result.ok, false)
  assert.equal(measured.result.error.code, expectedCode)
  assert.equal(measured.result.error.phase, 'StoreRelease')
  assert.deepEqual(measured.result.error.requested, [measured.event])
  assert.deepEqual(measured.result.error.prepared, scenario === 'duplicate' ? null : { durableEvents: [measured.event], cuts: [] })
  assert.equal(measured.causeIsInjected, true, 'same native Error identity remains inside the typed outcome')
}
