import assert from 'node:assert/strict'
import { createHash } from 'node:crypto'
import { execFileSync } from 'node:child_process'
import { readFileSync, writeFileSync, mkdtempSync, mkdirSync, rmSync } from 'node:fs'
import { join } from 'node:path'
import { tmpdir } from 'node:os'
import { assertBuildFresh } from '/Users/yuanxi/Workwork/vibe-fs/scripts/lib/build-state.mjs'
import {
  store, body, envelope, createdBody, work, transition, accepted,
  batch, append, current, mustOk,
} from '/Users/yuanxi/Workwork/vibe-fs/requirements/sphinx-v2/tests/persistence-support.mjs'

const root = '/Users/yuanxi/Workwork/vibe-fs'
const output = '/private/tmp/vibe-fs-b4-gen190-sealed-interpretations.json'
const sha256 = bytes => createHash('sha256').update(bytes).digest('hex')
const before = assertBuildFresh({ root })
assert.equal(before.generation, 190)
const manifestBytes = readFileSync(join(root, '.fable-build/build-manifest.json'))
const scenarios = []
const pending = body('InterpretationPending', {
  observationId: 'interpretation-observation', workId: 'interpretation-work', attempt: 1,
})
const applied = body('InterpretationApplied', {
  observationId: 'interpretation-observation', interpretationId: 'interpretation-receipt',
  pluginRef: 'proof-plugin@old', delta: envelope('{"fact":"opaque interpreted payload"}'),
})
const failed = body('InterpretationFailed', {
  observationId: 'interpretation-observation', interpretationId: 'interpretation-receipt',
  pluginRef: 'proof-plugin@old', reason: '真实旧解释异常\r\n原样保留',
})
const observation = accepted('interpretation-work', 'interpretation-observation')
observation.payload.canonicalResult = '{"answer":"原始回答\\r\\n保留"}'
const seed = [
  createdBody('实际旧 seal 捕获'),
  body('WorkPlanned', { work: [{ ...work('interpretation-work'), roundId: null }] }),
  transition('interpretation-work', 'Planned', {
    case: 'Running', fence: 'interpretation-work:1:logical', physicalRef: 'old-fixture-physical',
  }),
  observation,
  pending,
]

for (const [name, outcome] of [['pending', null], ['applied-noop', applied], ['failed-noop', failed]]) {
  const directory = mkdtempSync(join(tmpdir(), 'sphinx-v2-old-interpretation-capture-'))
  const commonDir = join(directory, '.git')
  mkdirSync(commonDir)
  const writerId = 'old-interpretation-' + name
  const writer = store.create(commonDir, writerId)
  let reopened
  try {
    const inquiry = 'legacy-interpretation-' + name
    const events = [await append(writer, batch(inquiry, 'first-transaction', seed))]
    if (outcome) events.push(await append(writer, batch(inquiry, 'old-outcome', [outcome], events[0])))
    const live = current(writer, inquiry)
    const state = mustOk(live)
    assert.equal(state.interpretations[0].value.status, 'pending')
    for (const event of events) assert.deepEqual(store.read(writer, event.id), event)
    assert.equal(live.stateHash, events.at(-1).payload.postStateFingerprint)
    const journalBytes = readFileSync(join(commonDir, 'wanxiang', 'events', writerId + '.ndjson'))
    store.dispose(writer)
    reopened = store.create(commonDir, writerId + '-cold-reader')
    assert.deepEqual(current(reopened, inquiry), live)
    scenarios.push({ name, inquiry, events, state, stateHash: live.stateHash,
      writerBytesBase64: journalBytes.toString('base64'), writerBytesSha256: sha256(journalBytes),
    })
  } finally {
    store.dispose(writer)
    if (reopened) store.dispose(reopened)
    rmSync(directory, { recursive: true, force: true })
  }
}

const after = assertBuildFresh({ root })
assert.deepEqual(after, before)
assert.equal(sha256(readFileSync(join(root, '.fable-build/build-manifest.json'))), sha256(manifestBytes))
const capture = {
  generation: before.generation,
  head: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: root, encoding: 'utf8' }).trim(),
  freshness: before,
  manifestSha256: sha256(manifestBytes),
  scenarios,
}
const bytes = Buffer.from(JSON.stringify(capture, null, 2) + '\n')
writeFileSync(output, bytes)
writeFileSync(output + '.sha256', sha256(bytes) + '\n')
writeFileSync('/private/tmp/vibe-fs-b4-gen190-capture-receipt.json', JSON.stringify({
  before, after, manifestSha256: capture.manifestSha256,
  fixturePath: output, fixtureSha256: sha256(bytes),
  scenarios: scenarios.map(value => ({ name: value.name, eventIds: value.events.map(event => event.id), stateHash: value.stateHash })),
}, null, 2) + '\n')
console.log(JSON.stringify({ fixture: output, sha256: sha256(bytes), generation: before.generation, scenarios: scenarios.length }))
