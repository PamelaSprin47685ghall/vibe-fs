import assert from 'node:assert/strict'
import test from 'node:test'
import { mkdtemp, rm } from 'node:fs/promises'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { inquiry, server } from './support/mcp.mjs'

test('WHAT[epistemic-reasoning-019] actual MCP restart restores results and rejects stale revision without changing durable export', async t => {
  const commonDir = await mkdtemp(join(tmpdir(), 'sphinx-generic-'))
  t.after(() => rm(commonDir, { recursive: true, force: true }))
  const result = { workId: 'work_restart_1', attempt: 1, payload: { text: 'accepted material' } }
  let inquiryId
  let before
  const first = await server(commonDir)
  try {
    const started = await first.tool('sphinx_inquiry_start', { question: 'durable generic probe' })
    assert.equal(started.isError, undefined)
    inquiryId = started.structuredContent.inquiryId
    const accepted = await first.tool('sphinx_work_submit', { inquiryId, expectedRevision: 0, results: [result] })
    assert.equal(accepted.isError, undefined)
    before = (await first.tool('sphinx_inquiry_export', { inquiryId })).structuredContent
    assert.equal(before.revision, 1)
    assert.deepEqual(before.results, [result])
  } finally {
    await first.close()
  }
  const second = await server(commonDir)
  try {
    const restored = await second.tool('sphinx_inquiry_export', { inquiryId })
    assert.deepEqual(restored.structuredContent, before)
    const stale = await second.tool('sphinx_work_submit', { inquiryId, expectedRevision: 0, results: [{ workId: 'stale' }] })
    assert.equal(stale.isError, true)
    assert.equal(stale._meta.error.code, 'REVISION_CONFLICT')
    assert.deepEqual((await second.tool('sphinx_inquiry_export', { inquiryId })).structuredContent, before)
    const next = await second.tool('sphinx_work_submit', {
      inquiryId, expectedRevision: 1, results: [{ ...result, workId: 'work_restart_2' }],
    })
    assert.equal(next.isError, undefined)
    assert.equal(next.structuredContent.revision, 2)
  } finally {
    await second.close()
  }
})

test('WHAT[epistemic-reasoning-019] actual generic submit rejects conflicting payload for an accepted work attempt', {
  todo: 'GAP-172: production generic Registry currently appends results without work identity conflict checks',
}, async () => {
  const current = await inquiry()
  const result = { workId: 'work_conflict', attempt: 1, payload: { text: 'first' } }
  assert.equal((await current.submit(0, [result])).isError, undefined)
  const before = (await current.export()).structuredContent
  const conflict = await current.submit(1, [{ ...result, payload: { text: 'changed' } }])
  assert.equal(conflict.isError, true)
  assert.deepEqual((await current.export()).structuredContent, before)
})
