import assert from 'node:assert/strict'
import test from 'node:test'
import { inquiry } from './support/mcp.mjs'

test('WHAT[epistemic-reasoning-034] actual MCP cancel refuses subsequent submissions at stale, current and future revisions', async () => {
  const current = await inquiry()
  const accepted = { workId: 'work_before', attempt: 1, payload: { text: 'already accepted' } }
  assert.equal((await current.submit(0, [accepted])).isError, undefined)
  const cancelled = await current.cancel()
  assert.equal(cancelled.isError, undefined)
  assert.equal(cancelled.structuredContent.status, 'cancelled')
  for (const expectedRevision of [0, 1, 2]) {
    const late = await current.submit(expectedRevision, [{ workId: 'work_late', attempt: 1 }])
    assert.equal(late.isError, true)
    assert.equal(late._meta.error.code, 'inquiry-cancelled')
  }
  const status = await current.status()
  assert.equal(status.isError, true)
  assert.equal(status._meta.error.code, 'inquiry-cancelled')
})

test('WHAT[epistemic-reasoning-034] actual durable cancellation survives a server restart and rejects late results', async t => {
  const { mkdtemp, rm } = await import('node:fs/promises')
  const { tmpdir } = await import('node:os')
  const { join } = await import('node:path')
  const { server } = await import('./support/mcp.mjs')
  const commonDir = await mkdtemp(join(tmpdir(), 'sphinx-cancel-'))
  t.after(() => rm(commonDir, { recursive: true, force: true }))
  const first = await server(commonDir)
  let inquiryId
  try {
    const started = await first.tool('sphinx_inquiry_start', { question: 'cancel probe' })
    inquiryId = started.structuredContent.inquiryId
    const cancelled = await first.tool('sphinx_inquiry_cancel', { inquiryId })
    assert.equal(cancelled.isError, undefined, JSON.stringify(cancelled))
    assert.equal(cancelled.structuredContent.status, 'cancelled')
  } finally {
    await first.close()
  }
  const second = await server(commonDir)
  try {
    const late = await second.tool('sphinx_work_submit', {
      inquiryId, expectedRevision: 0, results: [{ workId: 'work_late', attempt: 1 }],
    })
    assert.equal(late.isError, true)
    assert.equal(late._meta.error.code, 'inquiry-cancelled')
  } finally {
    await second.close()
  }
})
