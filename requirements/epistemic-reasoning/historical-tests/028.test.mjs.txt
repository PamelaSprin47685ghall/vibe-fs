import assert from 'node:assert/strict'
import test from 'node:test'
import { inquiry } from './support/mcp.mjs'

test('WHAT[epistemic-reasoning-028] actual MCP export preserves caller observation material without inventing source attribution', async () => {
  const current = await inquiry()
  const observations = [
    { workId: 'work_model', attempt: 1, payload: { claim: 'model belief', source: null } },
    { workId: 'work_source', attempt: 1, payload: { claim: 'document-backed material', source: { id: 'doc-1' } } },
  ]
  const submitted = await current.submit(0, observations)
  assert.equal(submitted.isError, undefined)
  const exported = await current.export()
  assert.equal(exported.isError, undefined)
  assert.deepEqual(exported.structuredContent.results, observations)
})

test('WHAT[epistemic-reasoning-028] actual export supplies canonical research bundle and classified claims', {
  todo: 'GAP-173: MCP export returns inquiry metadata and raw results, not the specified research bundle',
}, async () => {
  const current = await inquiry()
  const bundle = (await current.export()).structuredContent
  for (const field of ['events', 'eventHead', 'semanticHash', 'answerHash', 'answer', 'branchTree', 'claims']) {
    assert.ok(Object.hasOwn(bundle, field), `missing research field ${field}`)
  }
  assert.deepEqual(bundle.claims.filter(claim => claim.kind === 'externally-grounded-claim'), [])
})
