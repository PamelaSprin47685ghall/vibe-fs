import assert from 'node:assert/strict'
import test from 'node:test'
import { inquiry } from './support/mcp.mjs'

test('WHAT[epistemic-reasoning-033] actual generic submit does not repurchase or append an accepted work attempt', {
  todo: 'GAP-172: identical work attempt is currently appended again at the next revision',
}, async () => {
  const current = await inquiry()
  const observation = { workId: 'work_same', attempt: 1, payload: { text: 'accepted' } }
  assert.equal((await current.submit(0, [observation])).isError, undefined)
  const before = (await current.export()).structuredContent
  assert.equal((await current.submit(1, [structuredClone(observation)])).isError, undefined)
  assert.deepEqual((await current.export()).structuredContent, before)
})

test('WHAT[epistemic-reasoning-033] actual generic submit rejects work identities that were never issued', {
  todo: 'GAP-172: actual generic submit has no WorkItem or current-attempt admission gate',
}, async () => {
  const current = await inquiry()
  const before = (await current.export()).structuredContent
  const result = await current.submit(0, [{ workId: 'work_unissued', attempt: 999, payload: { text: 'late' } }])
  assert.equal(result.isError, true)
  assert.deepEqual((await current.export()).structuredContent, before)
})
